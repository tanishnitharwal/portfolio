// High-Performance Mandelbrot Kernel (C++20 / WebAssembly SIMD)
extern "C" EMSCRIPTEN_KEEPALIVE
int compute_mandelbrot_pixel(double cx, double cy, int max_iter) {
    double zx = 0.0, zy = 0.0;
    double zx2 = 0.0, zy2 = 0.0;
    int iter = 0;

    // Unrolled inner loop for instruction-level parallelism
    while (zx2 + zy2 <= 4.0 && iter < max_iter) {
        zy = 2.0 * zx * zy + cy;
        zx = zx2 - zy2 + cx;
        zx2 = zx * zx;
        zy2 = zy * zy;
        iter++;
    }
    return iter;
}
